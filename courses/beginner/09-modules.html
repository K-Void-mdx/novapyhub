Lesson 9: Modules and Imports

What are Modules?

Modules are Python files containing functions, classes, and variables that you can reuse. Python's standard library includes hundreds of modules for everything from math to web development.

Importing Modules

# Import the whole module
import math
print(math.pi)       # 3.141592653589793
print(math.sqrt(16)) # 4.0
print(math.floor(3.7))  # 3
print(math.ceil(3.2))   # 4

# Import specific items
from math import pi, sqrt
print(pi)           # 3.141592653589793
print(sqrt(25))     # 5.0

# Import with alias
import math as m
print(m.sin(m.pi/2))  # 1.0

# Import everything (not recommended)
from math import *

Useful Standard Library Modules

random — Random Numbers

import random

print(random.random())       # Random float 0.0 to 1.0
print(random.randint(1, 10))  # Random integer between 1 and 10
print(random.choice(["a", "b", "c"]))  # Random choice from list

# Shuffle a list
cards = ["A", "K", "Q", "J", "10"]
random.shuffle(cards)
print(cards)

# Generate a random password
import string
chars = string.ascii_letters + string.digits
password = ''.join(random.choice(chars) for _ in range(8))
print(password)

datetime — Dates and Times

from datetime import datetime, date, timedelta

# Current date and time
now = datetime.now()
print(now)          # 2026-07-30 15:30:00...
print(now.year)     # 2026
print(now.month)    # 7

# Formatting dates
print(now.strftime("%Y-%m-%d"))         # 2026-07-30
print(now.strftime("%B %d, %Y"))        # July 30, 2026

# Date arithmetic
today = date.today()
tomorrow = today + timedelta(days=1)
print(tomorrow)

os and sys — System Operations

import os
import sys

# Get current directory
print(os.getcwd())

# List files in directory
print(os.listdir("."))

# Create directory
# os.mkdir("new_folder")

# System info

# Command line arguments
# print(sys.argv)

json — Working with JSON

import json

# Convert Python to JSON string
data = {"name": "Alice", "age": 25, "city": "New York"}
json_string = json.dumps(data, indent=2)
print(json_string)

# Convert JSON string to Python
parsed = json.loads(json_string)
print(parsed["name"])  # Alice

Creating Your Own Module

Create a file my_utils.py:

# my_utils.py
def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

PI = 3.14159

Then use it in another file:

# main.py
import my_utils

print(my_utils.add(10, 5))      # 15
print(my_utils.PI)              # 3.14159

# Or
from my_utils import add, PI
print(add(3, 4))  # 7

The if __name__ Block

This pattern lets a file be both imported as a module AND run as a script:

# calculator.py
def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

if __name__ == "__main__":
    # This only runs when the file is executed directly
    print("Testing calculator:")
    print(f"5 + 3 = {add(5, 3)}")
    print(f"10 - 4 = {subtract(10, 4)}")
