Lesson 7: Dictionaries and Sets

What is a Dictionary?

A dictionary stores data in key-value pairs. Each key maps to a value — like a real dictionary where a word (key) has a definition (value).

# Creating dictionaries
student = {
    "name": "Alice",
    "age": 25,
    "grade": "A",
    "subjects": ["Math", "Physics"]
}

# Alternative creation
empty_dict = {}
grades = dict(alice=95, bob=87, charlie=92)

Accessing Dictionary Values

student = {"name": "Alice", "age": 25, "grade": "A"}

# Using square brackets
print(student["name"])     # "Alice"
# print(student["city"])   # KeyError!

# Using .get() (safe — returns None if missing)
print(student.get("name"))     # "Alice"
print(student.get("city"))     # None
print(student.get("city", "N/A"))  # "N/A" (default value)

Modifying Dictionaries

student = {"name": "Alice", "age": 25}

# Add or update
student["grade"] = "A"        # Add new key
student["age"] = 26           # Update existing

# Remove items
del student["grade"]          # Delete by key
age = student.pop("age")      # Remove and return value
student.clear()               # Remove all items

Dictionary Methods

student = {"name": "Alice", "age": 25, "grade": "A"}

# Get all keys
print(student.keys())    # dict_keys(['name', 'age', 'grade'])

# Get all values
print(student.values())  # dict_values(['Alice', 25, 'A'])

# Get all pairs
print(student.items())   # dict_items([('name', 'Alice'), ...])

# Loop through dictionary
for key, value in student.items():
    print(f"{key}: {value}")

# Check if key exists
print("name" in student)     # True
print("city" in student)     # False

# Merge dictionaries
a = {"x": 1, "y": 2}
b = {"y": 3, "z": 4}
a.update(b)
print(a)  # {'x': 1, 'y': 3, 'z': 4}

Dictionary Use Cases

# Counting items
text = "hello world"
count = {}
for char in text:
    if char in count:
        count[char] += 1
    else:
        count[char] = 1
print(count)
# {'h': 1, 'e': 1, 'l': 3, 'o': 2, ' ': 1, 'w': 1, 'r': 1, 'd': 1}

# Phone book
phone_book = {
    "Alice": "555-1234",
    "Bob": "555-5678",
    "Charlie": "555-9012"
}

# Nested dictionaries
users = {
    "alice123": {"name": "Alice", "age": 25, "email": "alice@email.com"},
    "bob456": {"name": "Bob", "age": 30, "email": "bob@email.com"}
}
print(users["alice123"]["email"])  # "alice@email.com"

What is a Set?

A set is an unordered collection of unique items. Use curly braces {} but without key-value pairs:

# Creating sets
fruits = {"apple", "banana", "orange", "apple"}
print(fruits)  # {'banana', 'orange', 'apple'} (no duplicates!)

# From a list
numbers = set([1, 2, 3, 2, 1])
print(numbers)  # {1, 2, 3}

# Empty set (not {} — that's a dict!)
empty = set()

Set Operations

a = {1, 2, 3, 4, 5}
b = {4, 5, 6, 7, 8}

# Union — all items from both sets
print(a | b)  # {1, 2, 3, 4, 5, 6, 7, 8}

# Intersection — items in both
print(a & b)  # {4, 5}

# Difference — items in a but not in b
print(a - b)  # {1, 2, 3}

# Symmetric difference — items in either but not both
print(a ^ b)  # {1, 2, 3, 6, 7, 8}

# Adding and removing
fruits = {"apple", "banana"}
fruits.add("orange")
fruits.remove("banana")
print(fruits)  # {'apple', 'orange'}

Set Use Cases

# Removing duplicates
names = ["Alice", "Bob", "Alice", "Charlie", "Bob"]
unique_names = list(set(names))
print(unique_names)  # ['Bob', 'Alice', 'Charlie']

# Finding common items
users_a = {"Alice", "Bob", "Charlie"}
users_b = {"Bob", "David", "Eve"}
common = users_a & users_b
print(common)  # {'Bob'}
