Lesson 8: Packaging and Deployment

Package your Python projects and deploy them to production.

Creating a Package

my_package/
    pyproject.toml
    src/
        my_package/
            __init__.py
            core.py

# pyproject.toml
[build-system]
requires = ["setuptools"]
build-backend = "setuptools.build_meta"

[project]
name = "my-package"
version = "0.1.0"
dependencies = ["requests>=2.0"]

# Install locally
pip install -e .

Docker

# Dockerfile
FROM python:3.12-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install -r requirements.txt
COPY . .
CMD ["python", "app.py"]

# Build & run
# docker build -t myapp .
# docker run myapp
