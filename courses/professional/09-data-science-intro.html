Lesson 9: Introduction to Data Science

NumPy, pandas, and matplotlib for data analysis.

NumPy

import numpy as np
arr = np.array([1, 2, 3, 4, 5])
print(arr.mean())     # 3.0
print(arr.sum())      # 15
print(arr * 2)        # [2, 4, 6, 8, 10]

matrix = np.array([[1, 2], [3, 4]])
print(matrix.T)  # Transpose

Pandas

import pandas as pd
df = pd.DataFrame({
    "name": ["Alice", "Bob", "Charlie"],
    "age": [25, 30, 35],
    "salary": [50000, 60000, 70000]
})
print(df.head())
print(df.describe())
print(df[df.age > 28])  # Filter
print(df.groupby("age").mean())

Matplotlib

import matplotlib.pyplot as plt
x = [1, 2, 3, 4, 5]
y = [2, 4, 6, 8, 10]
plt.plot(x, y)
plt.title("Simple Plot")
plt.xlabel("X")
plt.ylabel("Y")
plt.show()
