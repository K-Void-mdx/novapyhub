Lesson 3: Regular Expressions

Regular expressions (regex) are patterns for matching, searching, and manipulating text.

Basic Patterns

import re

# Match literal
re.search(r"Python", "I love Python")  # Match object

# Character classes
re.search(r"[aeiou]", "hello")     # 'e'
re.search(r"[0-9]", "abc123")      # '1'
re.search(r"[a-z]", "ABC")         # None

# Meta characters
.   # Any character except newline
^   # Start of string
$   # End of string
*   # 0 or more
+   # 1 or more
?   # 0 or 1
\d  # Digit [0-9]
\w  # Word character [a-zA-Z0-9_]
\s  # Whitespace

Common Functions

text = "Email: alice@email.com, Phone: 555-1234"

# search — find first match
email = re.search(r"\w+@\w+\.\w+", text)
print(email.group())  # alice@email.com

# findall — find all matches
phones = re.findall(r"\d{3}-\d{4}", text)
print(phones)  # ['555-1234']

# sub — replace
censored = re.sub(r"\w+@\w+\.\w+", "[HIDDEN]", text)
print(censored)  # Email: [HIDDEN], Phone: 555-1234

# split
parts = re.split(r"[,:]", text)
print(parts)

Groups

date_str = "2026-07-30"
pattern = r"(\d{4})-(\d{2})-(\d{2})"
match = re.search(pattern, date_str)
print(match.group(1))  # 2026
print(match.group(2))  # 07
print(match.group(3))  # 30

# Named groups
pattern = r"(?P<year>\d{4})-(?P<month>\d{2})"
match = re.search(pattern, date_str)
print(match.group("year"))   # 2026
print(match.group("month"))  # 07
