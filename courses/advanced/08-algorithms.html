Lesson 8: Algorithms and Complexity

Learn sorting, searching, recursion, dynamic programming, and Big O notation.

Binary Search

def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = (left + right) // 2
        if arr[mid] == target: return mid
        elif arr[mid] < target: left = mid + 1
        else: right = mid - 1
    return -1

print(binary_search([1, 3, 5, 7, 9], 5))  # 2

Bubble Sort

def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr

print(bubble_sort([64, 34, 25, 12, 22, 11, 90]))

Recursion: Fibonacci

def fib(n):
    if n <= 1: return n
    return fib(n-1) + fib(n-2)

# Memoized version
from functools import lru_cache
@lru_cache(maxsize=None)
def fib_fast(n):
    if n <= 1: return n
    return fib_fast(n-1) + fib_fast(n-2)

print(fib_fast(100))  # 354224848179261915075
