Lesson 5: Multithreading and Concurrency

Run multiple tasks concurrently with threads and processes.

Threading

import threading, time

def worker(name, delay):
    for i in range(3):
        print(f"{name}: {i}")
        time.sleep(delay)

threads = []
for name, delay in [("A", 0.5), ("B", 1), ("C", 0.3)]:
    t = threading.Thread(target=worker, args=(name, delay))
    threads.append(t)
    t.start()

for t in threads:
    t.join()

print("All done!")

Thread Safety with Lock

lock = threading.Lock()
counter = 0

def increment():
    global counter
    for _ in range(100000):
        with lock:
            counter += 1

concurrent.futures

from concurrent.futures import ThreadPoolExecutor
import time

def square(n):
    time.sleep(0.1)
    return n * n

with ThreadPoolExecutor(max_workers=4) as executor:
    results = list(executor.map(square, range(10)))
    print(results)  # [0, 1, 4, 9, ...]
