Lesson 3: File Handling

Reading and Writing Files

Python makes file operations simple. The built-in open() function is your gateway to file I/O.

Opening Files

# Different modes:
# "r" — read (default)
# "w" — write (overwrites!)
# "a" — append
# "r+" — read and write
# "b" — binary mode (add to another mode: "rb", "wb")

file = open("example.txt", "r")
content = file.read()
file.close()

The with Statement

Always use with to automatically close files:

# Read entire file
with open("example.txt", "r") as file:
    content = file.read()
    print(content)

# File is automatically closed when block exits

Reading Methods

# read all
with open("example.txt", "r") as f:
    content = f.read()
    print(content)

# read line by line
with open("example.txt", "r") as f:
    for line in f:
        print(line.strip())  # strip removes \n

# read all lines into list
with open("example.txt", "r") as f:
    lines = f.readlines()
    print(lines)  # ['line1\n', 'line2\n', ...]

# read specific number of chars
with open("example.txt", "r") as f:
    chunk = f.read(100)  # Read 100 characters

Writing to Files

# Write (overwrites if file exists)
with open("output.txt", "w") as f:
    f.write("Hello, World!\n")
    f.write("This is a new line.\n")

# Append (adds to existing content)
with open("output.txt", "a") as f:
    f.write("This line is appended.\n")

# Write multiple lines
lines = ["Line 1\n", "Line 2\n", "Line 3\n"]
with open("output.txt", "w") as f:
    f.writelines(lines)

Working with Different File Formats

CSV Files

import csv

# Reading CSV
with open("data.csv", "r") as f:
    reader = csv.reader(f)
    for row in reader:
        print(row)

# Writing CSV
with open("output.csv", "w", newline="") as f:
    writer = csv.writer(f)
    writer.writerow(["Name", "Age", "City"])
    writer.writerow(["Alice", 25, "NYC"])
    writer.writerow(["Bob", 30, "LA"])

JSON Files

import json

# Reading JSON
with open("data.json", "r") as f:
    data = json.load(f)
    print(data)

# Writing JSON
data = {
    "name": "Alice",
    "age": 25,
    "hobbies": ["reading", "coding"]
}
with open("output.json", "w") as f:
    json.dump(data, f, indent=2)

Checking if Files Exist

import os

if os.path.exists("example.txt"):
    print("File exists!")

if os.path.isfile("example.txt"):
    print("It's a file!")

if os.path.isdir("my_folder"):
    print("It's a directory!")

# Get file size
size = os.path.getsize("example.txt")
print(f"File size: {size} bytes")

Working with File Paths

import os

# Join paths safely
path = os.path.join("folder", "subfolder", "file.txt")
print(path)  # folder/subfolder/file.txt

# Get directory name
dir_name = os.path.dirname("/home/user/file.txt")
print(dir_name)  # /home/user

# Get file name
file_name = os.path.basename("/home/user/file.txt")
print(file_name)  # file.txt

# Split extension
name, ext = os.path.splitext("image.jpg")
print(f"Name: {name}, Extension: {ext}")  # Name: image, Extension: .jpg
