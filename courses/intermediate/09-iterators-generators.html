Lesson 9: Iterators and Generators Deep Dive

Custom iterators with __iter__ and __next__, generators with yield, and generator pipelines.

Custom Iterator

class CountDown:
    def __init__(self, n):
        self.n = n
    def __iter__(self):
        return self
    def __next__(self):
        if self.n <= 0:
            raise StopIteration
        self.n -= 1
        return self.n + 1

for x in CountDown(5):
    print(x)  # 5, 4, 3, 2, 1

Generator Functions

def fibonacci():
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b

fib = fibonacci()
print(next(fib))  # 0
print(next(fib))  # 1
print(next(fib))  # 1
print(next(fib))  # 2

Generator Pipelines

def read_file(filename):
    for line in open(filename):
        yield line.strip()

def filter_lines(lines, word):
    for line in lines:
        if word in line:
            yield line

pipeline = filter_lines(read_file("data.txt"), "Python")
for line in pipeline:
    print(line)
