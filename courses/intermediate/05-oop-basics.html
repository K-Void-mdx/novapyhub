Lesson 5: Object-Oriented Programming Basics

What is OOP?

Object-Oriented Programming is a paradigm that organizes code around objects (data + behavior) rather than functions and logic. Key concepts: classes, objects, attributes, methods, and inheritance.

Classes and Objects

A class is a blueprint. An object is an instance of that blueprint.

# Define a class
class Dog:
    # Class attribute (shared by all instances)
    species = "Canis familiaris"

    # Initializer (constructor)
    def __init__(self, name, age):
        self.name = name  # Instance attribute
        self.age = age    # Instance attribute

    # Instance method
    def bark(self):
        return f"{self.name} says Woof!"

    def description(self):
        return f"{self.name} is {self.age} years old"

# Create objects (instances)
dog1 = Dog("Buddy", 3)
dog2 = Dog("Max", 5)

print(dog1.name)         # Buddy
print(dog1.bark())       # Buddy says Woof!
print(dog2.description()) # Max is 5 years old
print(dog1.species)      # Canis familiaris

Instance Methods vs Class Methods

class Student:
    school = "Nova Py-Hub"

    def __init__(self, name, grade):
        self.name = name
        self.grade = grade

    # Instance method — uses self
    def info(self):
        return f"{self.name} - Grade: {self.grade}"

    # Class method — uses cls
    @classmethod
    def from_string(cls, data):
        name, grade = data.split(",")
        return cls(name.strip(), int(grade.strip()))

    # Static method — no self or cls
    @staticmethod
    def is_passing(grade):
        return grade >= 60

# Using class method as alternative constructor
s = Student.from_string("Alice, 85")
print(s.info())  # Alice - Grade: 85

Inheritance

# Parent class
class Animal:
    def __init__(self, name):
        self.name = name

    def speak(self):
        pass  # To be overridden

# Child class
class Cat(Animal):
    def speak(self):
        return f"{self.name} says Meow!"

class Dog(Animal):
    def speak(self):
        return f"{self.name} says Woof!"

animals = [Cat("Whiskers"), Dog("Buddy")]
for animal in animals:
    print(animal.speak())
